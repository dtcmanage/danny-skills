import heapq
def topo_order(nodes,edges):
 adj={n:set() for n in nodes}; deg={n:0 for n in nodes}
 for a,b in edges:
  if b not in adj[a]: adj[a].add(b); deg[b]+=1
 q=[n for n in nodes if not deg[n]]; heapq.heapify(q); out=[]
 while q:
  n=heapq.heappop(q); out.append(n)
  for m in adj[n]:
   deg[m]-=1
   if not deg[m]: heapq.heappush(q,m)
 if len(out)!=len(nodes): raise ValueError()
 return out
